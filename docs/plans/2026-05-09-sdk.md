# Agent Auth SDK Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construir el SDK embebible que los terceros (WhatsApp, Facebook, etc.) instalan en sus MCPs. Intercepta cada request del agente, corre el pipeline de validación llamando a la Platform API, y ejecuta la acción solo si todo está autorizado.

**Architecture:** Paquete npm en TypeScript. Pipeline de 5 funciones atómicas. HTTPS hardcodeado con verificación de certificado. Normalización de action y text antes de enviar a la API. Firmado con clave privada para verificación de integridad por los terceros.

**Tech Stack:** Node.js 18+, TypeScript, native `https` module (sin dependencias para TLS), zod, Jest

**Prerequisito:** Platform API (Plan 1) debe estar corriendo. Configurar `PLATFORM_API_URL` en el entorno.

```typescript
import https from 'https';
import http from 'http';
import { URL } from 'url';

export async function post<T>(url: string, body: unknown): Promise<T> {
  const parsed = new URL(url);

  if (parsed.protocol !== 'https:') {
    throw new Error('HTTPS required — HTTP is not allowed for security reasons');
  }

  return new Promise((resolve, reject) => {
    const data = JSON.stringify(body);

    const options: https.RequestOptions = {
      hostname: parsed.hostname,
      port: parsed.port || 443,
      path: parsed.pathname + parsed.search,
      method: 'POST',
      protocol: 'https:',
      rejectUnauthorized: true, // never skip TLS verification
      headers: {
        'Content-Type': 'application/json',
        'Content-Length': Buffer.byteLength(data),
      },
    };

    const req = https.request(options, (res) => {
      let raw = '';
      res.on('data', (chunk) => (raw += chunk));
      res.on('end', () => {
        try {
          resolve(JSON.parse(raw) as T);
        } catch {
          reject(new Error('Invalid JSON response from Platform API'));
        }
      });
    });

    req.on('error', reject);
    req.write(data);
    req.end();
  });
}
```

```typescript
export function normalizeText(text: string): string {
  if (!text) return '';

  let result = text;

  try {
    result = decodeURIComponent(result);

    // use original if decode fails
  }

  return result.normalize('NFKC').replace(ZERO_WIDTH_CHARS, '');
```

```typescript
import crypto from 'crypto';
import { post } from '../http/client';
import { normalizeAction } from '../normalize/action';
import { normalizeText } from '../normalize/text';
import { ValidationResponse } from '../types';

interface ValidateParams {
  apiKey: string;
  action: string;
  platform: string;
  text: string;
  platformApiUrl: string;
}

function hashKey(key: string): string {
  return crypto.createHash('sha256').update(key).digest('hex');
}

export async function validateKeyAndGetToken(params: ValidateParams): Promise<ValidationResponse> {
  const response = await post<ValidationResponse>(
    `${params.platformApiUrl}/v1/validate`,
    {
      api_key_hash: hashKey(params.apiKey),
      action: normalizeAction(params.action),
      platform: params.platform,
      text: normalizeText(params.text ?? ''),
    }
  );

  return response;
}
```

```typescript
export function verifyScope(action: string, scope: string[]): boolean {
```

- [ ] **Step 4: Implementar src/pipeline/verifyRules.ts**

```typescript
import { ValidationResponse } from '../types';

interface VerifyRulesParams {
  apiResponse: ValidationResponse;
}

interface RulesResult {
  blocked: boolean;
  error?: string;
}

export function verifyRules(params: VerifyRulesParams): RulesResult {
  if (!params.apiResponse.valid || params.apiResponse.error === 'action_not_permitted') {
    return { blocked: true, error: params.apiResponse.error };
  }
  return { blocked: false };
}
```

- [ ] **Step 3: Implementar src/index.ts**

```typescript
import { validateKeyAndGetToken } from './pipeline/validateKey';
import { verifyScope } from './pipeline/verifyScope';
import { verifyRules } from './pipeline/verifyRules';
import { AgentRequest, PipelineResult, SDKConfig } from './types';

export class AgentAuthSDK {
  private config: SDKConfig;

  constructor(config: SDKConfig) {
    if (!config.platformApiUrl.startsWith('https://')) {
      throw new Error('platformApiUrl must use HTTPS');
    }
    this.config = config;
  }

  async run(request: AgentRequest): Promise<PipelineResult> {
    // Función 1 + 2: Validar key y obtener token
    const apiResponse = await validateKeyAndGetToken({
      apiKey: request.apiKey,
      action: request.action,
      platform: request.platform,
      text: request.text ?? '',
      platformApiUrl: this.config.platformApiUrl,
    });

    if (!apiResponse.valid) {
      return { allowed: false, error: apiResponse.error ?? 'invalid_api_key' };
    }

    // Función 3: Verificar scope localmente
    if (!verifyScope(request.action, apiResponse.scope ?? [])) {
      return { allowed: false, error: 'action_not_permitted' };
    }

    // Función 4: Verificar reglas (resultado viene de la Platform API)
    const rulesResult = verifyRules({ apiResponse });
    if (rulesResult.blocked) {
      return { allowed: false, error: rulesResult.error ?? 'action_not_permitted' };
    }

    // Función 5: Aprobar
    return {
      allowed: true,
      token: apiResponse.token,
      userId: apiResponse.userId,
    };
  }
}

export type { AgentRequest, PipelineResult, SDKConfig } from './types';
```
